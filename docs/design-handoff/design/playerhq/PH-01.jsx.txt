function Treningstid({ empty, go }) {
  const { Button, AxisBadge } = window.PHQ.ns();
  const P = window.PHQ, U = window.PHQ_DATA.idag.uke;
  if (empty) return <Sec label="Treningstid denne uka"><Head k={"Treningstid uke " + U.week} meta="— AV — T" /><Muted>Uten treningsplan er det ingen fordeling å sammenligne med. Velg en plan, så ser du tid per akse mot det planen sier.</Muted><div><Button variant="secondary" icon="list-checks" onClick={() => go("PH-12")}>Velg treningsplan</Button></div></Sec>;
  const done = U.axes.reduce((a, [, h]) => a + (h || 0), 0), plan = U.axes.reduce((a, [, , g]) => a + g, 0);
  const rows = U.axes.map(([a, h, g]) => ({ a, h, g, sh: h == null ? null : Math.round(h / done * 100), ps: Math.round(g / plan * 100) }));
  const worst = rows.filter((r) => r.sh != null).sort((x, y) => Math.abs(y.sh - y.ps) - Math.abs(x.sh - x.ps))[0];
  const max = Math.max(...rows.map((r) => Math.max(r.g, r.h || 0))) * 1.1;
  return <Sec label="Treningstid denne uka">
    <Head k={"Treningstid uke " + U.week} meta={P.dec(done) + " AV " + P.dec(plan, 0) + " T"} />
    {worst && <p style={{ margin: 0, font: "500 15px/1.4 var(--font-sans)", color: "var(--text-primary)", textWrap: "pretty" }}>Du trener {worst.sh} % {AX[worst.a]}, planen sier {worst.ps} %.</p>}
    <div role="list" style={{ display: "flex", flexDirection: "column" }}>
      {rows.map((r) => <div role="listitem" key={r.a} aria-label={AX[r.a] + " " + P.dec(r.h) + " av " + P.dec(r.g) + " timer"} style={{ display: "grid", gridTemplateColumns: "64px minmax(0,1fr) auto", gap: 12, alignItems: "center", minHeight: 52, borderTop: "1px solid var(--border-hairline)" }}>
        <span><AxisBadge axis={r.a} size="sm" /></span>
        <div style={{ display: "flex", flexDirection: "column", gap: 6, minWidth: 0 }}>
          <div style={{ position: "relative", height: 8, background: "var(--surface-sunken)" }}><div style={{ height: "100%", width: ((r.h || 0) / max * 100) + "%", background: "var(--primary)" }}></div><span aria-hidden="true" style={{ position: "absolute", top: -4, bottom: -4, width: 2, left: "calc(" + (r.g / max * 100) + "% - 1px)", background: "var(--text-muted)" }}></span></div>
          <P.Meta>{r.sh == null ? "—" : r.sh + " %"} · PLAN {r.ps} %</P.Meta>
        </div>
        <P.Num s={{ font: "var(--type-num-s)", textAlign: "right" }}>{P.dec(r.h)} / {P.dec(r.g)} t</P.Num>
      </div>)}
    </div>
    <div style={{ display: "flex", flexDirection: "column", gap: 2, borderTop: "1px solid var(--border-hairline)", paddingTop: 8 }}><P.Meta>STREK = TIMER GJENNOMFØRT · MERKE = PLAN</P.Meta><P.Meta>{U.src}</P.Meta><P.Meta>{U.planSrc}</P.Meta></div>
  </Sec>;
}

function Popup({ go }) {
  const { Dialog, Button } = window.PHQ.ns();
  const P = window.PHQ, M = window.PHQ_DATA.idag.popup, K = "phq-popup-2609";
  const [open, setOpen] = React.useState(() => { try { return !sessionStorage.getItem(K); } catch (e) { return true; } });
  const close = () => { try { sessionStorage.setItem(K, "1"); } catch (e) {} setOpen(false); };
  return <Dialog open={open} title={M.kind} width={440} onClose={close} footer={<div style={{ display: "flex", gap: 8, flexWrap: "wrap", justifyContent: "flex-end", width: "100%" }}><Button variant="secondary" onClick={() => { close(); P.toast("Ligger i innboksen", "FINNES BAK BJELLA"); }}>Ignorer</Button><Button onClick={() => { close(); go("PH-21"); }}>Les</Button></div>}>
    <div style={{ display: "flex", flexDirection: "column", gap: 8 }}><P.Meta>{M.from.toUpperCase()} · {M.time}</P.Meta><span style={{ font: "600 17px/1.3 var(--font-sans)", color: "var(--text-primary)", textWrap: "pretty" }}>{M.title}</span><Muted>{M.text}</Muted></div>
  </Dialog>;
}
