const fmt = (v, enh) => v == null ? "—" : Number.isInteger(v) ? (v < 0 ? "−" + Math.abs(v) : String(v)) : window.PHQ.dec(v, Math.abs(v) < 1 ? 2 : 1);
function ML({ state, go }) {
  const { PageHeader, Button, ChoicePill, TextInput } = ns();
  const P = window.PHQ, I = window.IUP27, { cw, mob } = P.useW(), empty = state === "tom";
  const [rows, setRows] = useState(() => I.maltall.map((r) => empty ? { ...r, q: null, mal: null } : r));
  const [f, setF] = useState("alle"), [edit, setEdit] = useState(null), [val, setVal] = useState("");
  const vis = rows.filter((r) => f === "alle" || (f === "med" ? r.mal != null : r.mal == null));
  const groups = [...new Set(vis.map((r) => r.gruppe))];
  const commit = (r) => { const n = val.trim() === "" ? null : Number(val.replace(",", ".").replace("−", "-")); if (val.trim() !== "" && isNaN(n)) { P.toast("Skriv et tall", r.navn.toUpperCase()); return; } setRows((l) => l.map((x) => x.id === r.id ? { ...x, mal: n } : x)); setEdit(null); P.toast(n == null ? "Målet er fjernet" : "Målet er lagret", r.navn.toUpperCase() + " · " + fmt(n, r.enh) + " " + r.enh.toUpperCase()); };
  const malCell = (r) => edit === r.id ? <div style={{ display: "flex", gap: 6, alignItems: "center" }}><TextInput aria-label={"Mål " + r.navn} autoFocus inputMode="decimal" value={val} onChange={(e) => setVal(e.target.value)} onKeyDown={(e) => { if (e.key === "Enter") commit(r); if (e.key === "Escape") setEdit(null); }} style={{ width: 80, textAlign: "right" }} /><Button size="sm" icon="check" aria-label="Lagre mål" onClick={() => commit(r)}></Button></div>
    : <Button variant={r.mal == null ? "ghost" : "secondary"} size="sm" onClick={() => { setEdit(r.id); setVal(r.mal == null ? "" : String(r.mal).replace(".", ",")); }}>{r.mal == null ? "Sett mål" : fmt(r.mal, r.enh)}</Button>;
  const src = (r) => (r.type === "auto" ? "AUTO · " : "EGEN · ") + r.kilde.toUpperCase();
  const wide = cw >= 900, tpl = "minmax(0,1fr) repeat(4,56px) 64px 128px";
  return <P.Page max={1200}>

Årsresultat vises når alle fire kvartaler er målt." actions={<Button variant="secondary" icon="arrow-left" onClick={() => go("PH-11")}>Workbench</Button>} />

      {empty && <Muted>Ingen måltall er målt eller satt ennå. Auto-radene fylles når kildene har data.</Muted>}
      {groups.map((g) => <Sec key={g} k={g} meta={vis.filter((r) => r.gruppe === g).length + " MÅLTALL"}>
        {wide && <div style={{ display: "grid", gridTemplateColumns: tpl, gap: 8 }}>{["MÅLTALL", "K1", "K2", "K3", "K4", "ÅR", "MÅL 2026/27"].map((h, i) => <span key={h} style={{ textAlign: i && i < 6 ? "right" : "left" }}><P.Meta>{h}</P.Meta></span>)}</div>}
        {vis.filter((r) => r.gruppe === g).map((r) => { const q = r.q || [null, null, null, null], full = q.every((x) => x != null); return wide
          ? <div key={r.id} style={{ display: "grid", gridTemplateColumns: tpl, gap: 8, alignItems: "center", minHeight: 52, borderTop: "1px solid var(--border-hairline)" }}><span style={{ display: "flex", flexDirection: "column", gap: 2, minWidth: 0 }}><span style={{ font: "500 14px/1.3 var(--font-sans)", color: "var(--text-primary)" }}>{r.navn}</span><P.Meta>{r.enh.toUpperCase()} · {src(r)}</P.Meta></span>{q.map((x, i) => <P.Num key={i} s={{ textAlign: "right", color: x == null ? "var(--text-muted)" : undefined }}>{fmt(x, r.enh)}</P.Num>)}<P.Num s={{ textAlign: "right", color: "var(--text-muted)" }}>{full ? fmt(r.aar, r.enh) : "—"}</P.Num><span>{malCell(r)}</span></div>
          : <div key={r.id} style={{ display: "flex", flexDirection: "column", gap: 8, padding: "12px 0", borderTop: "1px solid var(--border-hairline)" }}><div style={{ display: "flex", gap: 8, alignItems: "center" }}><span style={{ flex: 1, display: "flex", flexDirection: "column", gap: 2, minWidth: 0 }}><span style={{ font: "500 14px/1.3 var(--font-sans)", color: "var(--text-primary)" }}>{r.navn}</span><P.Meta>{r.enh.toUpperCase()} · {src(r)}</P.Meta></span>{malCell(r)}</div><div style={{ display: "grid", gridTemplateColumns: "repeat(5,minmax(0,1fr))", gap: 4 }}>{[...q, full ? r.aar : null].map((x, i) => <span key={i} style={{ display: "flex", flexDirection: "column", gap: 2 }}><P.Meta>{i < 4 ?       </Sec>)}
    </P.Gate>
  </P.Page>;
}
